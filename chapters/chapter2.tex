\section{Ограниченность и непрерывность функции}

\begin{defn}
Функция $f$ называется \textit{ограниченной сверху (снизу) на множестве} $X\subset D_f$, если множество $f(X)$ ограничено сверху (снизу), т.е.~существует число~$M$ такое, что
$$
\forall x \in X \quad f(x) \le M \quad (\text{соотв., }f(x)\ge M).
$$
\end{defn}

Функция называется \textit{ограниченной на множестве}~$X$, если она на~$X$ ограничена и сверху, и снизу. Функция~$f$ называется \textit{ограниченной} (без указания множества), если она ограничена на всей области. Попросту говоря, функция ограниченна, если множество ее значений $f(D_f)$ ограничено.

\begin{defn}
Пусть функция $f$ определена на множестве $X\subset D_f$. Тогда 
$$
\sup_{X}\limits f \triangleq \sup f(X)\quad (\inf_{X}\limits f \triangleq \inf f(X))
$$ называется \textit{верхней (нижней) гранью числовой функции~$f$} на множестве~$X$. Также может обозначаться как $\sup_{x\in X}\limits f(x)$ (соотв., $\inf_{x\in X}\limits f(x)$)
\end{defn}

%----------------------------------------------------------------------------------------
% Определение 3.4 --- Непрерывность в точке
%----------------------------------------------------------------------------------------

\begin{defn} \label{df3:4}
Пусть функция $f$ определена в некоторой окрестности точки $a \in \bbR$. Тогда $f$ называется \textit{непрерывной в точке} $a$, если существует $\lim\limits_{x \to a} f(x) = f(a)$.

На языке Коши (определение предела 3.2) непрерывность $f$ в точке $a$ означает, что
$$
\forall\, \varepsilon > 0 \to \exists\, \delta > 0 \colon \quad \forall\, x, |x - a| < \delta \to |f(x) - f(a)| < \varepsilon.
$$
Оговорка $x \ne a$ (соответствующая неравенству $0 < |x - a| < \delta$) здесь уже не нужна, так как при $x = a$ имеем $|f(x) - f(a)| = 0 < \varepsilon$, и нужное неравенство выполняется автоматически.

На языке Гейне (определение предела 3.1) непрерывность $f$ в точке $a$ означает, что для любой последовательности $x_n$ такой, что $\lim\limits_{n \to \infty} x_n = a$, выполняется равенство $\lim\limits_{n \to \infty} f(x_n) = f(a)$. Оговорка $x_n \ne a$ здесь опять-таки не нужна. В самом деле, если $x_n$ принимает значения, равные $a$, то в последовательности $f(x_n)$ появляются члены, равные $f(a)$. Если их конечное число, то они ни на что не влияют. Если же их бесконечное число, то последовательность $f(x_n)$ приобретёт частичный предел, равный $f(a)$. Если и без этих членов $\lim\limits_{n \to \infty} f(x_n) = f(a)$, то новых частичных пределов не появится, предел останется равным $f(a)$. Если без этих членов утверждение $\lim\limits_{n \to \infty} f(x_n) = f(a)$ было неверным, то оно останется неверным и после учёта этих членов.
\end{defn}

%----------------------------------------------------------------------------------------
% Определение 3.5 --- Односторонняя непрерывность
%----------------------------------------------------------------------------------------

\begin{defn} \label{df3:5}
Пусть $a \in \bbR$ и функция $f$ определена в некоторой окрестности $a + 0$ (или $a - 0$). Тогда $f$ называется \textit{непрерывной справа} (соответственно \textit{слева}) в точке $a$, если существует $\lim\limits_{x \to a + 0} f(x) = f(a)$ (соответственно $\lim\limits_{x \to a - 0} f(x) = f(a)$).
\end{defn}

\noindent В дальнейшем будем применять следующие обозначения:
$$
\lim_{x \to a + 0} f(x) \equiv f(a + 0), \qquad \lim_{x \to a - 0} f(x) \equiv f(a - 0),
$$
$$
\lim_{x \to +\infty} f(x) \equiv f(+\infty), \qquad \lim_{x \to -\infty} f(x) \equiv f(-\infty).
$$

%----------------------------------------------------------------------------------------
% Определение 3.10 --- Непрерывность на промежутке
%----------------------------------------------------------------------------------------

\begin{defn} \label{df3:10}
Функция $f$, определённая на промежутке, называется непрерывной на этом промежутке, если она непрерывна во всех его внутренних точках, а в концах промежутка, если они принадлежат промежутку, имеет место соответствующая односторонняя непрерывность $f$.

Так, функция, определённая на отрезке $[a; b]$, называется непрерывной на $[a; b]$, если она непрерывна во всех точках интервала $(a; b)$, непрерывна справа в точке $a$ и непрерывна слева в точке $b$.
\end{defn}

\section{Теоремы Вейерштрасса}

%----------------------------------------------------------------------------------------
% Теорема 3.11 --- Первая теорема Вейерштрасса
%----------------------------------------------------------------------------------------

\begin{thm}[первая теорема Вейерштрасса] \label{th3:11}
Если функция $f$ непрерывна на отрезке $[a; b]$, то она ограничена на $[a; b]$.
\end{thm}

\begin{proof}
Пусть $f$ не является ограниченной на $[a; b]$. Тогда
$$
\forall\, E > 0 \to \exists\, x \in [a; b] \colon \quad |f(x)| > E.
$$
Возьмём $E = 1, 2, 3, \ldots, n, \ldots$ Тогда полученные значения $x$ образуют последовательность $x_n \in [a; b]$ такую, что при всех $n = 1, 2, \ldots$ выполняется неравенство $|f(x_n)| > n$. По теореме Больцано"--~Вейерштрасса можно выделить сходящуюся подпоследовательность $x_{n_k} \colon \lim\limits_{k \to \infty} x_{n_k} = x_0$. Так как все $x_{n_k} \in [a; b]$, то и $x_0 \in [a; b]$ (следствие из теоремы 2.2). Но $f$ непрерывна в точке $x_0$, значит, $\lim\limits_{k \to \infty} f(x_{n_k}) = f(x_0)$. Если $x_0$ "--- один из концов отрезка, например, $x_0 = a$, то $\lim\limits_{k \to \infty} x_{n_k} = a + 0$, а $f$ непрерывна справа в точке $a$. Равенство $\lim\limits_{k \to \infty} f(x_{n_k}) = f(x_0)$ сохраняется.

Но $|f(x_{n_k})| > n_k \ge k$ (так как $1 \le n_1 < n_2 < \ldots < n_k < \ldots$), поэтому $\lim\limits_{k \to \infty} f(x_{n_k}) = \infty$. Полученное противоречие показывает, что $f(x)$ ограничена на $[a; b]$.
\end{proof}

%----------------------------------------------------------------------------------------
% Теорема 3.12 --- Вторая теорема Вейерштрасса
%----------------------------------------------------------------------------------------

\begin{thm}[вторая теорема Вейерштрасса] \label{th3:12}
Если функция $f$ непрерывна на отрезке $[a; b]$, то существуют точки $x_1$ и $x_2$ на отрезке $[a; b]$ такие, что $f(x_1) = \sup\limits_{[a; b]} f(x)$, $f(x_2) = \inf\limits_{[a; b]} f(x)$ (функция достигает своих точных верхней и нижней граней).
\end{thm}

\begin{proof}
Докажем, что достигается $M = \sup\limits_{[a; b]} f(x)$ (для точной нижней грани доказательство аналогично).

По определению точной верхней грани, которая существует по теореме ~\ref{th3:11},
$$
(\forall\, x \in [a; b] \to f(x) \le M) \wedge
$$
$$
\wedge (\forall\, M' < M \to \exists\, x \in [a; b] \colon \quad f(x) > M').
$$
Рассмотрим $M' = M - \frac{1}{n}$, $n = 1, 2, \ldots$ Тогда полученные значения $x(M')$ образуют последовательность $x_n \in [a; b]$ такую, что при всех $n = 1, 2, \ldots$ выполняется неравенство
$$
M \ge f(x_n) > M - \frac{1}{n}.
$$
По теореме ~\ref{th2:3} $\lim\limits_{n \to \infty} f(x_n) = M$.

По теореме Больцано"--~Вейерштрасса можно выделить сходящуюся подпоследовательность $x_{n_k} \colon \lim\limits_{k \to \infty} x_{n_k} = x_0 \in [a; b]$. Но $f$ непрерывна в точке $x_0$, значит, $\lim\limits_{k \to \infty} f(x_{n_k}) = f(x_0)$. Случай, когда $x_0$ "--- один из концов отрезка, разбирается так же, как и в доказательстве теоремы ~\ref{th3:11}. С другой стороны, $\lim\limits_{k \to \infty} f(x_{n_k}) = M$. Значит, $M = f(x_0)$; $x_0$ и есть та точка, где достигается $\sup f(x)$.
\end{proof}

% Примеры 3.15 и 3.16 --- Непрерывность и ограниченность
%----------------------------------------------------------------------------------------

\noindent\textbf{Пример 3.15.} Функция, непрерывная на конечном интервале, не обязана быть ограниченной. Рассмотрим функцию $f(x) = \frac{1}{x}$ на $(0; 1)$. Она неограничена, так как
$$
\forall\, E > 0 \to \exists\, x(E) = \frac{1}{2E} \colon \quad f(x) > E.
$$

\noindent\textbf{Пример 3.16.} Функция, не являющаяся непрерывной на отрезке $[a; b]$, не обязана быть ограниченной. 

Рассмотрим функцию 
$f(x) = 
\begin{cases} 
\frac{1}{x}, & \text{если } x \in [-1; 0) \cup (0; 1]; \\ 
0, & \text{если } x = 0. 
\end{cases}$ \\

Она разрывна в точке $x = 0$ и неограничена на $[-1; 1]$ (аналогично примеру 3.15). График этой функции изображён на рис. 3.7. 

%----------------------------------------------------------------------------------------
% Примеры 3.17 и 3.18 --- Достижимость точных граней
%----------------------------------------------------------------------------------------

\noindent\textbf{Пример 3.17.} Функция, непрерывная на конечном интервале, может быть ограниченной, но не достигать ни точной верхней, ни точной нижней грани. Рассмотрим функцию $f(x) = x$ на $(0; 1)$. Она ограничена: $\sup\limits_{(0; 1)} f(x) = 1$, $\inf\limits_{(0; 1)} f(x) = 0$, ни одна из точных граней не достигается. 

\noindent\textbf{Пример 3.18.} Функция, не являющаяся непрерывной на отрезке $[a; b]$, может быть ограниченной, но не достигать ни точной верхней, ни точной нижней грани. Рассмотрим функцию 
$f(x) = 
\begin{cases} 
x, & \text{если } x \in (-1; 1); \\ 
0, & \text{если } x = \pm 1. 
\end{cases}$ 
Она разрывна в точках $x = \pm 1$ (в концах отрезка $[-1; 1]$ нет односторонней непрерывности); $\sup\limits_{x \in (-1; 1)} f(x) = 1$; $\inf\limits_{(-1; 1)} f(x) = -1$; ни одна из точных граней не достигается.
