\chapter{Теоремы о промежуточных значениях непрерывной функции.}
\section{Теорема о промежуточных значениях и теорема Больцано-Коши}

%----------------------------------------------------------------------------------------
% Теорема 3.14 --- О промежуточных значениях непрерывной функции
%----------------------------------------------------------------------------------------

\begin{thm}[о промежуточных значениях непрерывной функции] \label{th3:14}
Если функция $f$ непрерывна на отрезке $[a; b]$, то для любого значения $y_0$, заключённого между $f(a)$ и $f(b)$, существует точка $x_0 \in [a; b]$ такая, что $f(x_0) = y_0$.
\end{thm}

\begin{proof}
Если $y_0 = f(a)$ или $y_0 = f(b)$, то $x_0 = a$ или $x_0 = b$ соответственно. В противном случае рассмотрим функцию $g(x) = f(x) - y_0$. Тогда числа $g(a)$ и $g(b)$ имеют разный знак; по теореме Больцано"--~Коши найдётся точка $x_0 \in (a; b)$ такая, что $g(x_0) = 0$, т.е. $f(x_0) = y_0$.
\end{proof}

%----------------------------------------------------------------------------------------
% Следствие об образе отрезка
%----------------------------------------------------------------------------------------
\begin{cons}
    \textit{Множеством значений функции $f$, непрерывной на отрезке $[a; b]$, является отрезок $[m; M]$, где $m = \inf\limits_{[a; b]} f(x)$, $M = \sup\limits_{[a; b]} f(x)$.}
\end{cons} 

\begin{proof}
По теоремам ~\ref{th3:11} и ~\ref{th3:12} $m$ и $M$ "--- конечны, и найдутся точки $x_1$ и $x_2$ из отрезка $[a; b]$ такие, что $f(x_1) = m$; $f(x_2) = M$. Рассмотрим $f$ на отрезке $[x_1; x_2]$ (или $[x_2; x_1]$, смотря что больше). По теореме ~\ref{th3:14}
$$
\forall\, y_0 \in [m; M] \to \exists\, x_0 \in [x_1; x_2] \colon \quad f(x_0) = y_0.
$$
Значит, $y_0 \in E(f)$, но $E(f) \subset [m; M]$, поэтому $E(f) = [m; M]$.
\end{proof}

%----------------------------------------------------------------------------------------
% Теорема 3.13 --- Больцано-Коши (о нуле непрерывной функции)
%----------------------------------------------------------------------------------------

\begin{thm}[Больцано"--~Коши] \label{th3:13}
Если функция $f$ непрерывна на отрезке $[a; b]$ и принимает в точках $a$ и $b$ значения разного знака (т.е. $f(a)f(b) < 0$), то существует точка $c \in (a; b)$ такая, что $f(c) = 0$.
\end{thm}

%----------------------------------------------------------------------------------------
% Пример 3.19 --- Невыполнимость теоремы Больцано-Коши для разрывной функции
%----------------------------------------------------------------------------------------

\begin{exmpl} \label{exmpl:19}
    Для разрывных на $[a; b]$ функций теорема не обязана выполняться. Рассмотрим функцию
$$
f(x) = 
\begin{cases} 
1, & \text{если } 0 \le x \le 1; \\ 
-1, & \text{если } -1 \le x < 0. 
\end{cases}
$$
На концах отрезка $[-1; 1]$ функция принимает значения разных знаков, но нигде на $(-1; 1)$ не обращается в нуль.

\end{exmpl}
